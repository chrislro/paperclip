<objective>
Stand up the server-side authoritative storage for per-user clinical config. After this plan, the backend has:

1. A `user_configs` table (one row per user) holding rx_templates (JSON), voices (JSON), custom_instructions (text), doctor_name (text), and three boolean toggles (auto_cid, auto_clear_soap, auto_transcribe_on_dictation).

class UserConfig(Base):
    """Per-user clinical configuration: prescription templates, SOAP voices,
    custom instructions, doctor name, and behavioral toggles. One row per User.

    Storage strategy: rx_templates and voices are JSON columns (lists of dicts).
    Server is authoritative; client treats `chrome.storage.local[config_<uid>]`
    as a read-through cache wiped on signout.

    Lazy creation: a row is created by GET /api/me/config on first access for a
    new user, seeded with DEFAULT_RX_TEMPLATES.
    """
    __tablename__ = "user_configs"

    user_id = Column(Integer, ForeignKey("users.id"), primary_key=True)
    rx_templates = Column(JSON, nullable=False, default=list)
    voices = Column(JSON, nullable=False, default=list)
    custom_instructions = Column(Text, nullable=False, default="")
    doctor_name = Column(String(255), nullable=False, default="")
    auto_cid = Column(Boolean, nullable=False, default=True)
    auto_clear_soap = Column(Boolean, nullable=False, default=True)
    auto_transcribe_on_dictation = Column(Boolean, nullable=False, default=False)
    created_at = Column(DateTime, default=datetime.utcnow, nullable=False)
    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow, nullable=False)
```

def _serialize_user_config(cfg):
    """Shape the UserConfig row for JSON response. Mirrors PATCH input shape."""
    return {
        "rx_templates": cfg.rx_templates or [],
        "voices": cfg.voices or [],
        "custom_instructions": cfg.custom_instructions or "",
        "doctor_name": cfg.doctor_name or "",
        "auto_cid": bool(cfg.auto_cid),
        "auto_clear_soap": bool(cfg.auto_clear_soap),
        "auto_transcribe_on_dictation": bool(cfg.auto_transcribe_on_dictation),
        "updated_at": cfg.updated_at.isoformat() if cfg.updated_at else None,
    }


def _get_or_seed_user_config(session, user_id):
    """Return the user's config row, creating + seeding it if absent."""
    cfg = session.query(UserConfig).filter_by(user_id=user_id).first()
    if cfg is None:
        cfg = UserConfig(
            user_id=user_id,
            rx_templates=[dict(t) for t in DEFAULT_RX_TEMPLATES],
            voices=[],
            custom_instructions="",
            doctor_name="",
            auto_cid=True,
            auto_clear_soap=True,
            auto_transcribe_on_dictation=False,
        )
        session.add(cfg)
        session.commit()
    return cfg


@bp.route("/api/me/config", methods=["GET"])
@require_auth
def api_me_config_get():
    """Return the authenticated user's clinical config.

    Lazy-seeds DEFAULT_RX_TEMPLATES on first access for a new user.
    """
    from emr_automation.database import get_session
    session = get_session()
    cfg = _get_or_seed_user_config(session, g.user_id)
    return jsonify(_serialize_user_config(cfg))


@bp.route("/api/me/config", methods=["PATCH"])
@require_auth
def api_me_config_patch():
    """Partial-update the authenticated user's clinical config.

    Accepts any subset of: rx_templates (list), voices (list),
    custom_instructions (str), doctor_name (str),
    auto_cid / auto_clear_soap / auto_transcribe_on_dictation (bool).

    Last-writer-wins by updated_at. No optimistic-concurrency check —
    clinical config is single-author per user.
    """
    from emr_automation.database import get_session
    session = get_session()
    data = request.get_json(silent=True)
    if not isinstance(data, dict):
        return jsonify({"error": "JSON object body required"}), 400

    cfg = _get_or_seed_user_config(session, g.user_id)

    if "rx_templates" in data:
        if not isinstance(data["rx_templates"], list):
            return jsonify({"error": "rx_templates must be an array"}), 400
        cfg.rx_templates = data["rx_templates"]

    if "voices" in data:
        if not isinstance(data["voices"], list):
            return jsonify({"error": "voices must be an array"}), 400
        cfg.voices = data["voices"]

    if "custom_instructions" in data:
        cfg.custom_instructions = str(data["custom_instructions"])[:5000]

    if "doctor_name" in data:
        cfg.doctor_name = str(data["doctor_name"])[:255]

    for k in ("auto_cid", "auto_clear_soap", "auto_transcribe_on_dictation"):
        if k in data:
            setattr(cfg, k, bool(data[k]))

    cfg.updated_at = datetime.utcnow()
    session.commit()
    return jsonify(_serialize_user_config(cfg))
```
